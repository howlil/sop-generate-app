### Step 4: Apply CHECK Constraints (Raw SQL)

**File:** `server/prisma/constraints.sql`

**Content:**
```sql
-- [P1-A] XOR constraint: RiwayatTandaTangan
ALTER TABLE RiwayatTandaTangan ADD CONSTRAINT chk_tte_xor
  CHECK (
    (sopDetailId IS NOT NULL AND pengajuanEvaluasiId IS NULL)
    OR
    (sopDetailId IS NULL AND pengajuanEvaluasiId IS NOT NULL)
  );

-- [P1-B] nilaiOPD hanya untuk TERJADWAL
ALTER TABLE PengajuanEvaluasi ADD CONSTRAINT chk_nilai_opd_only_terjadwal
  CHECK (
    (jenis = 'MANDIRI' AND nilaiOPD IS NULL)
    OR (jenis = 'TERJADWAL')
  );

-- [P2-A] TERMINATOR/TASK/DECISION constraint
ALTER TABLE LangkahSOP ADD CONSTRAINT chk_langkah_cabang
  CHECK (
    (jenis = 'TERMINATOR' AND langkahSelanjutnyaYaId IS NULL AND langkahSelanjutnyaTidakId IS NULL)
    OR (jenis = 'TASK' AND langkahSelanjutnyaTidakId IS NULL)
    OR (jenis = 'DECISION')
  );
```

### Step 5: Create Triggers (Raw SQL)

**File:** `server/prisma/triggers.sql`

**Content:**
```sql
-- [P0-B] Satu versi BERLAKU per SOP
DELIMITER $$
CREATE TRIGGER trg_satu_berlaku_per_sop
BEFORE UPDATE ON DetailSOP
FOR EACH ROW
BEGIN
  IF NEW.status = 'BERLAKU' AND OLD.status != 'BERLAKU' THEN
    IF EXISTS (
      SELECT 1 FROM DetailSOP
      WHERE sopId = NEW.sopId
        AND status = 'BERLAKU'
        AND id != NEW.id
    ) THEN
      SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'SOP sudah memiliki versi BERLAKU. Tetapkan versi lama ke DIGANTIKAN dulu.';
    END IF;
  END IF;
END$$
DELIMITER ;

-- [P0-D] Status terminal guard (DIGANTIKAN/DICABUT tidak bisa diubah)
DELIMITER $$
CREATE TRIGGER trg_detailsop_status_terminal
BEFORE UPDATE ON DetailSOP
FOR EACH ROW
BEGIN
  IF NEW.status != OLD.status AND OLD.status IN ('DIGANTIKAN', 'DICABUT') THEN
    SIGNAL SQLSTATE '45000'
      SET MESSAGE_TEXT = 'Status terminal SOP tidak bisa diubah';
  END IF;
END$$
DELIMITER ;
```

        nomorSOP: 'TEST-002',
        logoInstansi: 'logo',
        namaLembaga: 'Test',

    } else {
      console.log('❌ FAILED: Wrong error:', e.message);

  }

  // Test [P2-D]: 1 KEPALA_OPD per OPD
  console.log('\nTest [P2-D]: 1 KEPALA_OPD per OPD');
  try {

## Rollback Plan

**If migration fails:**

```bash
# 1. Drop all tables
mysql -u root -p sop_biro_organisasi -e "
SET FOREIGN_KEY_CHECKS = 0;
SHOW TABLES;
SET FOREIGN_KEY_CHECKS = 1;
"

# Or manually:
mysql -u root -p sop_biro_organisasi
DROP TABLE IF EXISTS LogEditSOP, RiwayatTandaTangan, ...;  # All 20 tables

# 2. Reset Prisma migrations
cd server
rm -rf prisma/migrations
npx prisma migrate dev --name init_schema_20_models

# 3. If still failing, check MariaDB logs
# /var/log/mysql/error.log atau docker logs
```

**If CHECK constraints fail:**
```bash
# Remove constraints and re-apply
mysql -u root -p sop_biro_organisasi
ALTER TABLE RiwayatTandaTangan DROP CONSTRAINT IF EXISTS chk_tte_xor;
# ... repeat for other constraints

# Then re-apply constraints.sql
```

**If triggers fail:**
```bash
# Remove triggers
mysql -u root -p sop_biro_organisasi
DROP TRIGGER IF EXISTS trg_satu_berlaku_per_sop;
DROP TRIGGER IF EXISTS trg_detailsop_status_terminal;

# Then re-apply triggers.sql
```
